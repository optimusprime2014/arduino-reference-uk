= bool

// OVERVIEW SECTION STARTS
[#overview]
--

[float]
=== Опис
У тип `bool` можна записати одне з двох значень: `link:../../constants/truefalse[true]` або `link:../../constants/truefalse[false]`. (Кожна змінна `bool` займає один байт пам'яті.)


[%hardbreaks]


[float]
=== Синтаксис
`bool var = val;`


[float]
=== Параметри
`var`: ім'я змінної. +
`val`: значення, яке присвоюється цій змінній.

=== Приклад коду
// Describe what the example code is all about and add relevant code   ►►►►► THIS SECTION IS MANDATORY ◄◄◄◄◄
Цей код показує, як використовувати тип даних `bool`.

[source,arduino]
----
int LEDpin = 5;     // LED на піні №5
int switchPin = 13; // кнопка на піні №13

bool running = false;

void setup() {
  pinMode(LEDpin, OUTPUT);
  pinMode(switchPin, INPUT);
  digitalWrite(switchPin, HIGH);  // встановлюємо внутрішній підтягуючий резистор для кнопки
}

void loop() {
  if (digitalRead(switchPin) == LOW) {
    // кнопка натиснута, перемикаємо стан змінної running
    delay(100);                     // затримка для уникнення "дебаунсу / брязкання контактів" кнопки
    running = !running;             // перемикаємо значення змінної running
    digitalWrite(LEDpin, running);  // встановлюємо стан світлодіода відповідно до змінної running
  }
}
----
